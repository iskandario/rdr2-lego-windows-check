cmake_minimum_required(VERSION 3.21)
project(AssassinTraversal VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(traversal_tests tests/TraversalTests.cpp)
if(MSVC)
  set_property(TARGET traversal_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(traversal_tests PRIVATE /W4 /WX /UNDEBUG)
else()
  target_compile_options(traversal_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME traversal COMMAND traversal_tests)
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Only Windows x64 is supported")
  endif()
  set(RDR2_SDK "" CACHE PATH "Directory containing SDK/inc and SDK/lib")
  if(NOT EXISTS "${RDR2_SDK}/inc/natives.h" OR NOT EXISTS "${RDR2_SDK}/lib/ScriptHookRDR2.lib")
    message(FATAL_ERROR "Set RDR2_SDK to the ScriptHookRDR2 SDK directory")
  endif()
  add_library(AssassinTraversal SHARED src/Plugin.cpp)
  target_include_directories(AssassinTraversal SYSTEM PRIVATE "${RDR2_SDK}/inc")
  target_link_libraries(AssassinTraversal PRIVATE "${RDR2_SDK}/lib/ScriptHookRDR2.lib")
  target_compile_options(AssassinTraversal PRIVATE /W3 /utf-8)
  set_target_properties(AssassinTraversal PROPERTIES PREFIX "" SUFFIX ".asi"
    MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
